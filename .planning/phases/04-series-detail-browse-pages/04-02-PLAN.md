**File 4: `src/components/series/series-detail-content.tsx`** (Client component)

```typescript
'use client'
import { useState } from 'react'
import { SeasonSelector } from './season-selector'
import { EpisodeList } from './episode-list'
import type { Season, Episode, Show } from '@/lib/mock-data'

interface SeriesDetailContentProps {
  seasons: Season[]
  episodesBySeason: Record<string, Episode[]>
  showBrandColor: Show['brandColor']
}

export function SeriesDetailContent({ seasons, episodesBySeason, showBrandColor }: SeriesDetailContentProps) {
  const [activeSeasonId, setActiveSeasonId] = useState(seasons[0]?.id ?? '')
  const currentEpisodes = episodesBySeason[activeSeasonId] ?? []

  return (
    <>
      {seasons.length > 0 && (
        <SeasonSelector
          seasons={seasons}
          activeSeason={activeSeasonId}
          onSeasonChange={setActiveSeasonId}
        />
      )}
      <div className="mt-6">
        <EpisodeList episodes={currentEpisodes} showBrandColor={showBrandColor} />
      </div>
    </>
  )
}
```

Then in page.tsx, import `SeriesDetailContent` and pass `seasons`, `episodesBySeason`, `showBrandColor`.
