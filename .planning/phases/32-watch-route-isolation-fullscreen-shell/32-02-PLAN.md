Switch sequence (strict order from RESEARCH.md Pattern 3):
```typescript
function handleEpisodeSwitch(newEpisode: ViewEpisode) {
  if (!playerRef.current || !newEpisode.youtubeId) return
  forceFlush()                        // 1. save outgoing position FIRST (closure has old episodeId)
  setCurrentEpisode(newEpisode)       // 2. trigger React rebuild → useProgressTracker rebuilds save() with new episodeId
  playerRef.current.loadVideoById({   // 3. swap video (no iframe remount, auto-plays)
    videoId: newEpisode.youtubeId,
    startSeconds: 0,                  // always 0 — switching is intentional navigation, not resume
  })
  window.history.replaceState(        // 4. URL sync without re-render
    null,
    '',
    `/${locale}/watch/${newEpisode.id}`
  )
}
```
