# Plan 19-01: Editorial Header

## Objective

Replace the current glassmorphic pill-shaped navbar with the editorial-style header: yellow dot + "CLICKIT" text logo, nav links for authenticated users (Browse, Coming Soon, Support), anchor nav on landing (Why Clickit, Shows, FAQ), Sign in + Start free buttons for unauthenticated visitors, user avatar when signed in. Sticky with backdrop blur, transparent on landing, solid on scroll. No search bar.

Rewrite `src/components/layout/navbar.tsx` entirely. The new header:

```tsx
'use client'

import { useState, useEffect } from 'react'
import { useTranslations } from 'next-intl'
import { usePathname } from '@/i18n/navigation'
import { Link } from '@/i18n/navigation'
import { cn } from '@/lib/utils'
import { AuthSection } from './auth-section'

interface NavbarProps {
  initialUser?: { id: string; email: string } | null
}

export function Navbar({ initialUser }: NavbarProps) {
  const t = useTranslations('nav')
  const pathname = usePathname()
  const [scrolled, setScrolled] = useState(false)
  const isLanding = pathname === '/' || pathname === ''

  useEffect(() => {
    let ticking = false
    const handler = () => {
      if (!ticking) {
        requestAnimationFrame(() => {
          setScrolled(window.scrollY > 24)
          ticking = false
        })
        ticking = true
      }
    }
    window.addEventListener('scroll', handler, { passive: true })
    return () => window.removeEventListener('scroll', handler)
  }, [])

  return (
    <header
      className={cn(
        'fixed inset-x-0 top-0 z-50 flex items-center gap-7 px-10 py-[18px] border-b transition-all duration-200',
        'backdrop-blur-[18px]',
        isLanding && !scrolled
          ? 'bg-gradient-to-b from-[rgba(10,10,10,0.8)] to-transparent border-transparent'
          : '',
        isLanding && scrolled
          ? 'bg-[rgba(10,10,10,0.96)] border-[var(--line)]'
          : '',
        !isLanding && !scrolled
          ? 'bg-[linear-gradient(180deg,rgba(10,10,10,0.95),rgba(10,10,10,0.65))] border-[var(--line)]'
          : '',
        !isLanding && scrolled
          ? 'bg-[rgba(10,10,10,0.96)] border-[var(--line)]'
          : '',
      )}
      style={{ WebkitBackdropFilter: 'blur(18px)' }}
    >
      {/* Logo: yellow dot + CLICKIT text */}
      <Link
        href="/"
        className="flex items-center gap-2 font-[family-name:var(--font-ui)] text-xl font-bold tracking-[-0.03em] text-[var(--text)]"
      >
        <span className="inline-block h-[7px] w-[7px] rounded-full bg-[var(--primary)] mb-0.5" />
        CLICKIT
      </Link>

      {/* Nav links */}
      <nav className="flex items-center gap-[22px]">
        {!isLanding && (
          <>
            <Link href="/browse" className={cn('text-[13.5px] text-[var(--text-dim)] py-1.5 px-0.5 relative transition-colors duration-150 hover:text-[var(--text)]', pathname === '/browse' && 'text-[var(--text)] after:absolute after:inset-x-0 after:bottom-[-4px] after:h-0.5 after:bg-[var(--primary)]')}>
              Browse
            </Link>
            <Link href="/coming-soon" className={cn('text-[13.5px] text-[var(--text-dim)] py-1.5 px-0.5 relative transition-colors duration-150 hover:text-[var(--text)]', pathname === '/coming-soon' && 'text-[var(--text)] after:absolute after:inset-x-0 after:bottom-[-4px] after:h-0.5 after:bg-[var(--primary)]')}>
              Coming Soon
            </Link>
            <Link href="/support" className={cn('text-[13.5px] text-[var(--text-dim)] py-1.5 px-0.5 relative transition-colors duration-150 hover:text-[var(--text)]', pathname === '/support' && 'text-[var(--text)] after:absolute after:inset-x-0 after:bottom-[-4px] after:h-0.5 after:bg-[var(--primary)]')}>
              Support
            </Link>
          </>
        )}
        {isLanding && (
          <>
            <a href="#features" className="text-[13.5px] text-[var(--text-dim)] py-1.5 px-0.5 transition-colors duration-150 hover:text-[var(--text)]">Why Clickit</a>
            <a href="#shows" className="text-[13.5px] text-[var(--text-dim)] py-1.5 px-0.5 transition-colors duration-150 hover:text-[var(--text)]">Shows</a>
            <a href="#faq" className="text-[13.5px] text-[var(--text-dim)] py-1.5 px-0.5 transition-colors duration-150 hover:text-[var(--text)]">FAQ</a>
          </>
        )}
      </nav>

      {/* Right side: auth */}
      <div className="ms-auto flex items-center gap-3.5">
        <AuthSection initialUser={initialUser} isLanding={isLanding} />
      </div>
    </header>
  )
}
```

Key changes from current:
- Removed pill-shaped container — header is now full-width
- Removed `<Image>` logo → text "CLICKIT" with yellow dot
- Added nav links (Browse, Coming Soon, Support for auth'd; anchor links for landing)
- No search bar (HEAD-06)
- Transparent on landing, solid on scroll
- Active nav link gets yellow underline via `after:` pseudo
- Uses `usePathname()` from `@/i18n/navigation` for landing detection and active link highlighting
</action>
