```typescript
// ---------------------------------------------------------------------------
// Catalog stage query
// ---------------------------------------------------------------------------

export type CatalogStage = 'zero' | 'growing' | 'full'

export const getCatalogStage = unstable_cache(
  async (): Promise<{ stage: CatalogStage; showCount: number }> => {
    const supabase = createPublicClient()
    const { count } = await supabase
      .from('shows')
      .select('*', { count: 'exact', head: true })
      .eq('archived', false)
      .in('status', ['now-streaming', 'coming-soon'])

    const showCount = count ?? 0
    const stage: CatalogStage =
      showCount === 0 ? 'zero' :
      showCount <= 5 ? 'growing' :
      'full'

    return { stage, showCount }
  },
  ['catalog-stage'],
  { revalidate: 3600, tags: ['shows'] }
)
```

Key details per D-04:
- Uses `createPublicClient()` (NOT `createClient()`) -- safe for unstable_cache closure (no cookies)
- Uses `{ count: 'exact', head: true }` -- returns count header only, zero row data transferred
- Revalidate 3600s (1 hour), tags `['shows']` -- matches existing show query patterns
- `createPublicClient` import already exists at top of file (line 3)
