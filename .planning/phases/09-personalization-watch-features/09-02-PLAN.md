export async function getEpisodeProgress(
  userId: string,
  episodeId: string
): Promise<{ progressSeconds: number; durationSeconds: number; completed: boolean } | null> {
  const supabase = await createClient()
  const { data } = await supabase
    .from('watch_progress')
    .select('progress_seconds, duration_seconds, completed')
    .eq('user_id', userId)
    .eq('episode_id', episodeId)
    .maybeSingle()

  if (!data) return null
  return {
    progressSeconds: data.progress_seconds,
    durationSeconds: data.duration_seconds ?? 0,
    completed: data.completed,
  }
}

export async function getContinueWatchingEpisodes(
  userId: string
): Promise<ContinueWatchingItem[]> {
  const supabase = await createClient()
  const { data } = await supabase
    .from('watch_progress')
    .select('progress_seconds, duration_seconds, episode_id, episodes!inner(id, title, episode_number, show_id, season_id, seasons!inner(season_number), shows!inner(id, title, slug, brand_color))')
    .eq('user_id', userId)
    .eq('completed', false)
    .gt('progress_seconds', 0)
    .order('updated_at', { ascending: false })
    .limit(10)

  if (!data) return []

  return data.map((row: any) => {
    const ep = row.episodes
    const dur = row.duration_seconds || 1
    return {
      episodeId: ep.id,
      episodeTitle: ep.title,
      episodeNumber: ep.episode_number,
      seasonNumber: ep.seasons.season_number,
      showId: ep.shows.id,
      showTitle: ep.shows.title,
      showSlug: ep.shows.slug,
      showBrandColor: (ep.shows.brand_color || 'brand-blue') as ViewShow['brandColor'],
      progressSeconds: row.progress_seconds,
      durationSeconds: dur,
      progressPercent: Math.min(Math.round((row.progress_seconds / dur) * 100), 99),
    }
  })
}

```typescript
'use client'

import { useEffect, useRef, useCallback } from 'react'
import { createClient } from '@/lib/supabase/client'

export function useProgressTracker(
  player: YT.Player | null,
  episodeId: string,
  userId: string | null
) {
  const lastSavedRef = useRef<number>(0)
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null)

  const flush = useCallback(() => {
    if (!player || !userId) return
    try {
      const currentTime = Math.floor(player.getCurrentTime())
      const duration = Math.floor(player.getDuration())
      // Skip saves below 2 seconds to avoid overwriting with post-seek 0 values
      if (currentTime < 2 && lastSavedRef.current > 2) return
      if (Math.abs(currentTime - lastSavedRef.current) < 10) return

      const completed = currentTime / duration >= 0.95
      const supabase = createClient()
      supabase
        .from('watch_progress')
        .upsert(
          {
            user_id: userId,
            episode_id: episodeId,
            progress_seconds: currentTime,
            duration_seconds: duration,
            completed,
          },
          { onConflict: 'user_id,episode_id' }
        )
        .then(() => {
          lastSavedRef.current = currentTime
        })
    } catch {
      // Player may be destroyed; swallow error
    }
  }, [player, episodeId, userId])

  // Force flush (used for ended/pause events) - bypasses the 10-second delta check
  const forceFlush = useCallback(() => {
    if (!player || !userId) return
    try {
      const currentTime = Math.floor(player.getCurrentTime())
      const duration = Math.floor(player.getDuration())
      if (currentTime < 2 && lastSavedRef.current > 2) return

      const completed = currentTime / duration >= 0.95
      const supabase = createClient()
      supabase
        .from('watch_progress')
        .upsert(
          {
            user_id: userId,
            episode_id: episodeId,
            progress_seconds: currentTime,
            duration_seconds: duration,
            completed,
          },
          { onConflict: 'user_id,episode_id' }
        )
        .then(() => {
          lastSavedRef.current = currentTime
        })
    } catch {
      // Swallow
    }
  }, [player, episodeId, userId])

  // Start 10-second interval
  useEffect(() => {
    if (!player || !userId) return

    intervalRef.current = setInterval(flush, 10_000)

    return () => {
      if (intervalRef.current) clearInterval(intervalRef.current)
    }
  }, [player, userId, flush])

  // Visibility change handler
  useEffect(() => {
    if (!player || !userId) return

    const handleVisibility = () => {
      if (document.hidden) forceFlush()
    }

    document.addEventListener('visibilitychange', handleVisibility)
    return () => document.removeEventListener('visibilitychange', handleVisibility)
  }, [player, userId, forceFlush])

  // Flush on unmount (route change)
  useEffect(() => {
    return () => { forceFlush() }
  }, [forceFlush])

  return { flush, forceFlush }
}
```

'use client'

import { useState, useEffect, useRef } from 'react'

interface AutoplayOverlayProps {
  nextEpisode: {
    id: string
    title: string
    seasonNumber: number
    episodeNumber: number
  }
  onPlayNow: () => void
  onCancel: () => void
}

export function AutoplayOverlay({
  nextEpisode,
  onPlayNow,
  onCancel,
}: AutoplayOverlayProps) {
  const [countdown, setCountdown] = useState(8)
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null)

  useEffect(() => {
    intervalRef.current = setInterval(() => {
      setCountdown((prev) => {
        if (prev <= 1) {
          if (intervalRef.current) clearInterval(intervalRef.current)
          onPlayNow()
          return 0
        }
        return prev - 1
      })
    }, 1000)

    return () => {
      if (intervalRef.current) clearInterval(intervalRef.current)
    }
  }, [onPlayNow])

  return (
    <div className="absolute inset-0 z-20 flex items-end justify-between bg-gradient-to-t from-black/90 via-black/50 to-transparent p-4 md:p-6 max-sm:flex-col max-sm:items-stretch max-sm:gap-4">
      {/* Left column - next episode info */}
      <div className="flex flex-col gap-2">
        <span className="text-xs font-medium text-white/50 uppercase tracking-widest">
          Up next
        </span>
        <span className="text-sm text-white/70">
          S{nextEpisode.seasonNumber} E{nextEpisode.episodeNumber}
        </span>
        <span className="text-lg font-semibold text-white">
          {nextEpisode.title}
        </span>
        <span className="inline-flex items-center gap-1.5 rounded-full bg-white/10 px-3 py-1 text-xs text-white/70 w-fit">
          Playing in {countdown}s
        </span>
      </div>

      {/* Right column - action buttons */}
      <div className="flex flex-col gap-2 items-end max-sm:flex-row max-sm:items-stretch max-sm:w-full">
        <button
          onClick={onPlayNow}
          className="rounded-full bg-white text-black px-6 py-2 text-sm font-semibold hover:bg-white/90 transition-colors duration-200 max-sm:flex-1"
        >
          Play now
        </button>
        <button
          onClick={() => {
            if (intervalRef.current) clearInterval(intervalRef.current)
            onCancel()
          }}
          className="rounded-full border border-white/20 text-white/70 px-4 py-2 text-sm hover:border-white/40 hover:text-white transition-all duration-200 max-sm:flex-1"
        >
          Cancel
        </button>
      </div>
    </div>
  )
}
