```typescript
'use client'

import { useState, useEffect, useRef, useCallback } from 'react'

interface UseIdleTimerReturn {
  isIdle: boolean
  resetTimer: () => void
}

export function useIdleTimer(delay: number): UseIdleTimerReturn {
  const [isIdle, setIsIdle] = useState(false)
  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)

  const resetTimer = useCallback(() => {
    setIsIdle(false)
    if (timerRef.current) clearTimeout(timerRef.current)
    timerRef.current = setTimeout(() => setIsIdle(true), delay)
  }, [delay])

  useEffect(() => {
    resetTimer()
    return () => {
      if (timerRef.current) clearTimeout(timerRef.current)
    }
  }, [resetTimer])

  return { isIdle, resetTimer }
}
```

CRITICAL rules:
- Use bare `setTimeout` / `clearTimeout` — NOT `window.setTimeout`. TypeScript narrows `window.setTimeout` to `never` inside guard blocks. This is a confirmed project pitfall (MEMORY.md).
- `ReturnType<typeof setTimeout>` — not `number`, not `NodeJS.Timeout`.
- `resetTimer` wrapped in `useCallback([delay])` — ensures stable ref across renders.
- `isIdle` starts `false` — chrome is visible immediately on mount, then hides after first 3s.
