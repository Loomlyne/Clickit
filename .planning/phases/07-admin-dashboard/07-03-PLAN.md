```typescript
interface YouTubeUrlInputProps {
  value: string              // Current URL or video ID
  onChange: (videoId: string | null, rawUrl: string) => void
  error?: string
}
```

This component wraps a text input field. On every `onChange`:
1. Store the raw URL in local state for display
2. Call `extractYouTubeId(rawUrl)` from `@/lib/youtube`
3. If valid ID extracted: call `props.onChange(videoId, rawUrl)`, show thumbnail preview below input
4. If no valid ID and input is not empty: show validation error "Could not extract video ID from this URL"
5. If empty: call `props.onChange(null, '')`

Thumbnail preview (when valid ID):
- `<img src={getYouTubeThumbnailUrl(videoId)} alt="Video thumbnail" />` at 120x68px
- Video ID displayed in `text-muted-foreground text-sm` below the thumbnail
- Container: flex row, gap-3, items-center

Input field:
- Label: "YouTube URL" (handled by parent form, not this component)
- Placeholder: `Paste YouTube URL (e.g., youtube.com/watch?v=...)`
- Full-width text input

For edit mode: if `value` prop is a video ID (11 chars, no slashes), display the thumbnail immediately. If it's a full URL, extract and display.

**Step 3: Create show detail page (D-01, D-02).**
Create `src/app/admin/shows/[id]/page.tsx` as a Server Component:

```typescript
import { createClient } from '@/lib/supabase/server'
import { notFound } from 'next/navigation'
import Link from 'next/link'
import { SeasonTabs } from '@/components/admin/season-tabs'
import { Button } from '@/components/ui/button'

export default async function ShowDetailPage({ params }: { params: Promise<{ id: string }> }) {
  const { id } = await params
  const supabase = await createClient()

  // Fetch show
  const { data: show } = await supabase
    .from('shows')
    .select('*')
    .eq('id', id)
    .single()

  if (!show) notFound()

  // Fetch seasons ordered by season_number
  const { data: seasons } = await supabase
    .from('seasons')
    .select('*')
    .eq('show_id', id)
    .order('season_number', { ascending: true })

  // Fetch all episodes for this show, ordered by episode_number
  const { data: episodes } = await supabase
    .from('episodes')
    .select('*')
    .eq('show_id', id)
    .eq('archived', false)
    .order('episode_number', { ascending: true })

  // Group episodes by season_id
  const episodesBySeason: Record<string, typeof episodes> = {}
  for (const ep of episodes ?? []) {
    if (!episodesBySeason[ep.season_id]) {
      episodesBySeason[ep.season_id] = []
    }
    episodesBySeason[ep.season_id].push(ep)
  }

  return (
    <div>
      {/* Breadcrumb */}
      <nav className="text-sm text-muted-foreground mb-4">
        <Link href="/admin" className="hover:text-foreground">Shows</Link>
        <span className="mx-2">&gt;</span>
        <span className="text-foreground">{show.title}</span>
      </nav>

      {/* Page header */}
      <div className="flex items-center justify-between mb-6">
        <h1 className="text-xl font-semibold">{show.title}</h1>
        {/* Edit Show button opens the show form drawer -- handled by a client wrapper */}
      </div>

      {/* Season tabs with episodes */}
      <SeasonTabs
        seasons={seasons ?? []}
        episodesBySeason={episodesBySeason}
        showId={id}
      />
    </div>
  )
}
```
