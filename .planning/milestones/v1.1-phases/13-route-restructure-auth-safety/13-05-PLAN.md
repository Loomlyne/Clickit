```tsx
'use client'

import { useTransition } from 'react'
import { Check } from 'lucide-react'
import { cn } from '@/lib/utils'
import { useRouter, usePathname } from '@/i18n/navigation'
import type { Locale } from '@/i18n/routing'

const options: Array<{ locale: Locale; label: string; sublabel: string }> = [
  { locale: 'en', label: 'English', sublabel: 'English' },
  { locale: 'ar', label: 'العربية', sublabel: 'Arabic' },
]

/**
 * URL-based locale switcher. Replaces the former cookie-based version.
 *
 * Click flow:
 *   1. usePathname() returns the current path WITHOUT the locale prefix
 *      (e.g., /profile — not /en/profile).
 *   2. router.replace(pathname, { locale: next }) re-navigates to the same
 *      path with the next locale prepended.
 *   3. No cookie write, no revalidatePath — the URL is the source of truth.
 *      next-intl automatically sets NEXT_LOCALE as a cookie on successful
 *      locale-prefixed navigation, so other code paths (e.g., the OAuth
 *      callback in plan 13-04) can still read it.
 *
 * useTransition hides any latency between the click and the re-navigation
 * so the active state doesn't flicker during concurrent rendering.
 */
export function LanguagePreference({ current }: { current: Locale }) {
  const router = useRouter()
  const pathname = usePathname()
  const [isPending, startTransition] = useTransition()

  function select(next: Locale) {
    if (next === current || isPending) return
    startTransition(() => {
      router.replace(pathname, { locale: next })
    })
  }

  return (
    <div className="flex gap-3">
      {options.map(({ locale, label, sublabel }) => {
        const active = locale === current
        return (
          <button
            key={locale}
            type="button"
            onClick={() => select(locale)}
            disabled={isPending}
            className={cn(
              'relative flex-1 rounded-[16px] border p-4 text-left transition-all duration-200 disabled:opacity-60',
              active
                ? 'border-white/30 bg-white/[0.08]'
                : 'border-white/[0.08] bg-white/[0.02] hover:border-white/[0.16] hover:bg-white/[0.04]'
            )}
          >
            {active && (
              <span className="absolute right-3 top-3 flex h-5 w-5 items-center justify-center rounded-full bg-white">
                <Check className="h-3 w-3 text-black" strokeWidth={2.5} />
              </span>
            )}
            <p className="text-sm font-semibold text-white">{label}</p>
            <p className="mt-0.5 text-xs text-white/40">{sublabel}</p>
          </button>
        )
      })}
    </div>
  )
}
```

```ts
import type { MetadataRoute } from 'next'

import { getStreamingShows, getComingSoonShows } from '@/lib/queries'
import { siteUrl } from '@/lib/site-url'
import { routing } from '@/i18n/routing'

/**
 * Sitemap generator. Emits one entry per route with `alternates.languages`
 * pointing at the /en/ and /ar/ variants. Google uses the alternates to
 * treat the two URLs as language variants of the same canonical page
 * instead of competing duplicates.
 *
 * Episode URLs are intentionally omitted — the catalog has ~300 episodes
 * addressed by numeric IDs that won't help crawlers discover content. Show
 * pages already link to every episode via the season tabs, so crawlers
 * reach them via depth-2 traversal. Revisit if organic search becomes a
 * meaningful traffic source.
 */
function languageAlternates(base: string, path: string): Record<string, string> {
  // Build { en: ${base}/en${path}, ar: ${base}/ar${path} } — one entry per
  // locale in routing.locales. Adding a new locale in routing.ts picks up
  // automatically here.
  return Object.fromEntries(
    routing.locales.map((loc) => [loc, `${base}/${loc}${path === '/' ? '' : path}`]),
  )
}

function entry(
  base: string,
  path: string,
  now: Date,
  changeFrequency: NonNullable<MetadataRoute.Sitemap[number]['changeFrequency']>,
  priority: number,
): MetadataRoute.Sitemap[number] {
  // The `url` field uses the defaultLocale variant as the canonical URL the
  // crawler hits first. Google follows alternates to discover the /ar/ peer.
  const defaultPath = path === '/' ? '' : path
  return {
    url: `${base}/${routing.defaultLocale}${defaultPath}`,
    lastModified: now,
    changeFrequency,
    priority,
    alternates: {
      languages: languageAlternates(base, path),
    },
  }
}

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const base = siteUrl()
  const [streaming, comingSoon] = await Promise.all([
    getStreamingShows(),
    getComingSoonShows(),
  ])

  const now = new Date()

  const staticEntries: MetadataRoute.Sitemap = [
    entry(base, '/', now, 'daily', 1.0),
    entry(base, '/browse', now, 'daily', 0.9),
    entry(base, '/coming-soon', now, 'weekly', 0.5),
  ]

  const showEntries: MetadataRoute.Sitemap = [...streaming, ...comingSoon].map((show) =>
    entry(base, `/series/${show.slug}`, now, 'weekly', 0.8),
  )

  return [...staticEntries, ...showEntries]
}
```
