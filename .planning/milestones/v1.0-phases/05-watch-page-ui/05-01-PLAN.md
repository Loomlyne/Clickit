```typescript
/** Resolve full episode context from episode ID (Phase 5) */
export function getEpisodeById(episodeId: string): {
  episode: Episode
  show: Show
  season: Season
  seasons: Season[]
  episodesBySeason: Record<string, Episode[]>
} | undefined {
  const episode = MOCK_EPISODES.find((e) => e.id === episodeId)
  if (!episode) return undefined
  const show = MOCK_SHOWS.find((s) => s.id === episode.showId)
  if (!show) return undefined
  const season = MOCK_SEASONS.find((s) => s.id === episode.seasonId)
  if (!season) return undefined
  const seasons = getSeasonsByShowId(show.id)
  const episodesBySeason: Record<string, Episode[]> = {}
  for (const s of seasons) {
    episodesBySeason[s.id] = getEpisodesBySeasonId(s.id)
  }
  return { episode, show, season, seasons, episodesBySeason }
}

/** Compute prev/next published episodes across season boundaries (Phase 5) */
export function getEpisodeNavigation(
  episodeId: string,
  seasons: Season[],
  episodesBySeason: Record<string, Episode[]>
): { prev: Episode | null; next: Episode | null } {
  const allEpisodes = seasons.flatMap(
    (season) => (episodesBySeason[season.id] ?? []).filter((ep) => ep.status === 'published')
  )
  const currentIndex = allEpisodes.findIndex((ep) => ep.id === episodeId)
  if (currentIndex === -1) return { prev: null, next: null }
  return {
    prev: currentIndex > 0 ? allEpisodes[currentIndex - 1] : null,
    next: currentIndex < allEpisodes.length - 1 ? allEpisodes[currentIndex + 1] : null,
  }
}
```

**3. Create `src/components/watch/youtube-player.tsx`:**

Client component (`'use client'`). Implements per D-01, D-02, D-03, D-04.

Module-level singleton pattern for YouTube IFrame API script loading:
- `let apiReady = false`, `let apiLoading = false`, `const readyCallbacks: (() => void)[] = []`
- `loadYouTubeAPI()` returns `Promise<void>`, creates script tag `https://www.youtube.com/iframe_api`, registers `window.onYouTubeIframeAPIReady`
- Handles multiple callers via `readyCallbacks` array

Component `YouTubePlayer` with props:
```typescript
interface YouTubePlayerProps {
  videoId: string
  brandColor?: string // Tailwind bg class for loading state (e.g., 'bg-brand-purple')
  onReady?: (player: YT.Player) => void
  onStateChange?: (state: number) => void
  onError?: (errorCode: number) => void
}
```

Implementation:
- `containerRef = useRef<HTMLDivElement>(null)`, `playerRef = useRef<YT.Player | null>(null)`
- Store callback props in refs to avoid useEffect re-triggering (per RESEARCH.md Pitfall 5)
- `useEffect` depends only on `[videoId]`: calls `loadYouTubeAPI()`, creates `new YT.Player(containerRef.current, {...})` with playerVars: `{ autoplay: 0, controls: 1, rel: 0, playsinline: 1, enablejsapi: 1, iv_load_policy: 3, origin: window.location.origin }`. Do NOT include `modestbranding`, `showinfo`, `autohide`, or `theme` (all deprecated per RESEARCH.md).
- Cleanup: `playerRef.current?.destroy(); playerRef.current = null`
- `cancelled` flag pattern to prevent stale initialization
- `useState<boolean>(false)` for `isReady` to swap loading state for player
- `useState<number | null>(null)` for `errorCode` to show error fallback

Loading state (shown while `!isReady && !errorCode`):
- `<div className={cn("w-full aspect-video min-h-[200px] flex items-center justify-center", brandColor ?? "bg-surface-2")}>`
- Centered `<Play className="size-12 text-white/20" />` icon from lucide-react

Error state (shown when `errorCode` is set):
- Same container dimensions (`w-full aspect-video min-h-[200px]`)
- `bg-surface-2` background
- `role="alert"` on the message container
- Error 101 or 150: "This video is currently unavailable. Try again later."
- Error 100: "Video not found. It may have been removed."
- Other errors: "Something went wrong loading this video. Please refresh the page."
- Text: `text-sm text-muted-foreground` centered

Player container:
- Outer: `<div role="region" aria-label="Video player" className="w-full aspect-video min-h-[200px] bg-black">`
- Inner: `<div ref={containerRef} className="w-full h-full" />` -- YouTube replaces this with iframe
- The loading/error states are conditionally rendered; the actual player div is always mounted but hidden with `className={cn("w-full h-full", !isReady && "hidden")}` so the iframe initializes in the background
