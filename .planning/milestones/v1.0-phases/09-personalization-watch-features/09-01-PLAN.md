```typescript
// ---------------------------------------------------------------------------

export async function getWatchlistShows(userId: string): Promise<ViewShow[]> {
  const supabase = await createClient()
  const { data } = await supabase
    .from('watchlist')
    .select('show_id, created_at, shows(*)')
    .eq('user_id', userId)
    .order('created_at', { ascending: false })

  if (!data) return []
  return data
    .map((row: any) => row.shows)
    .filter(Boolean)
    .map(mapShow)
}

export async function getShowWatchlistState(userId: string, showId: string): Promise<boolean> {
  const supabase = await createClient()
  const { data } = await supabase
    .from('watchlist')
    .select('show_id')
    .eq('user_id', userId)
    .eq('show_id', showId)
    .maybeSingle()

  return !!data
}

export async function getShowsWatchlistState(userId: string, showIds: string[]): Promise<Set<string>> {
  if (showIds.length === 0) return new Set()
  const supabase = await createClient()
  const { data } = await supabase
    .from('watchlist')
    .select('show_id')
    .eq('user_id', userId)
    .in('show_id', showIds)

  return new Set((data ?? []).map((row) => row.show_id))
}
```

```typescript
'use client'

import { useState, useCallback } from 'react'
import { useRouter } from 'next/navigation'
import { toast } from 'sonner'
import { createClient } from '@/lib/supabase/client'

interface UseWatchlistReturn {
  isInList: boolean
  toggle: () => void
  isLoading: boolean
}

export function useWatchlist(
  showId: string,
  initialSaved: boolean,
  userId: string | null
): UseWatchlistReturn {
  const [isInList, setIsInList] = useState(initialSaved)
  const [isLoading, setIsLoading] = useState(false)
  const router = useRouter()

  const toggle = useCallback(() => {
    if (!userId) {
      router.push(`/auth?next=${encodeURIComponent(window.location.pathname)}`)
      return
    }

    const prev = isInList
    setIsInList(!prev)
    setIsLoading(true)

    const supabase = createClient()
    const mutation = prev
      ? supabase.from('watchlist').delete().eq('user_id', userId).eq('show_id', showId)
      : supabase.from('watchlist').insert({ user_id: userId, show_id: showId })

    mutation.then(({ error }) => {
      if (error) {
        setIsInList(prev)
        toast.error(prev ? 'Failed to remove from list' : 'Failed to add to list')
      }
      setIsLoading(false)
    })
  }, [showId, userId, isInList, router])

  return { isInList, toggle, isLoading }
}
```

```typescript
'use client'

import { Plus, Check } from 'lucide-react'
import { cn } from '@/lib/utils'
import { useWatchlist } from '@/hooks/use-watchlist'

interface WatchlistToggleButtonProps {
  showId: string
  variant: 'card' | 'hero'
  initialSaved?: boolean
  userId: string | null
}

export function WatchlistToggleButton({
  showId,
  variant,
  initialSaved = false,
  userId,
}: WatchlistToggleButtonProps) {
  const { isInList, toggle, isLoading } = useWatchlist(showId, initialSaved, userId)

  const Icon = isInList ? Check : Plus

  if (variant === 'card') {
    return (
      <button
        onClick={(e) => {
          e.preventDefault()
          e.stopPropagation()
          toggle()
        }}
        disabled={isLoading}
        aria-label={isInList ? 'Remove from My List' : 'Add to My List'}
        className={cn(
          'absolute right-3 top-3 z-20 flex h-9 w-9 items-center justify-center rounded-full transition-all duration-200',
          isInList
            ? 'bg-white text-black border border-white'
            : 'border border-white/20 bg-black/50 backdrop-blur-md text-white/80 hover:bg-black/70'
        )}
      >
        <Icon className="size-4" />
      </button>
    )
  }

  // Hero variant
  return (
    <button
      onClick={toggle}
      disabled={isLoading}
      aria-label={isInList ? 'Remove from My List' : 'Add to My List'}
      className={cn(
        'flex h-11 w-11 items-center justify-center rounded-full border backdrop-blur-sm transition-colors',

          ? 'bg-white text-black border-white'

  )
}
```

```typescript
import type { Metadata } from 'next'
import { redirect } from 'next/navigation'
import Link from 'next/link'
import { createClient } from '@/lib/supabase/server'
import { getWatchlistShows } from '@/lib/queries'
import { PortraitShowCard } from '@/components/shows/portrait-show-card'
import { SectionHeader } from '@/components/primitives/section-header'
import { ScrollReveal } from '@/components/primitives/scroll-reveal'

export const metadata: Metadata = {
  title: 'My List | Clickit Story',
  description: 'Your saved shows on Clickit Story',
}

export default async function MyListPage() {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()

  if (!user) redirect('/auth?next=/my-list')

  const shows = await getWatchlistShows(user.id)

  return (
    <div className="pb-16">
      {/* Page header */}
      <div className="relative py-20 px-6 md:px-12 text-center overflow-hidden">
        <div
          className="absolute inset-0 pointer-events-none"
          style={{
            background:
              'radial-gradient(ellipse 60% 80% at 50% 100%, rgba(44,53,105,0.15), transparent)',
          }}
        />
        <div className="relative">
          <p
            className="text-sm tracking-[0.3em] uppercase mb-4"
            style={{ color: 'var(--color-brand-blue-safe)' }}
          >
            My List
          </p>
          <h1 className="font-display font-bold text-3xl sm:text-4xl md:text-5xl text-white leading-tight">
            Saved Shows
          </h1>
          <p className="text-white/50 text-lg mt-3 max-w-md mx-auto">
            {shows.length > 0
              ? `${shows.length} show${shows.length !== 1 ? 's' : ''} in your list`
              : 'Save shows to watch them later.'}
          </p>
        </div>
      </div>

      {/* Show grid or empty state */}
      {shows.length > 0 ? (
        <div className="px-6 md:px-12">
          <SectionHeader title="Your Shows" />
          <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 xl:grid-cols-5 2xl:grid-cols-6 gap-4 md:gap-5 mt-2">
            {shows.map((show, i) => (
              <ScrollReveal key={show.id} delay={i * 0.04}>

        <div className="flex flex-col items-center justify-center gap-4 py-32 text-center">

        </div>

  )
```
